    startBtn:document.getElementById('start-btn'),

let audioCtx;

function beep(f,d,t='sine',v=0.15){

    try{

        if(!audioCtx) audioCtx=new (window.AudioContext||window.webkitAudioContext)();

        const o=audioCtx.createOscillator(),g=audioCtx.createGain();

        o.connect(g);g.connect(audioCtx.destination);o.type=t;o.frequency.value=f;

        g.gain.setValueAtTime(v,audioCtx.currentTime);

        g.gain.exponentialRampToValueAtTime(0.001,audioCtx.currentTime+d);

        o.start();o.stop(audioCtx.currentTime+d);

    }catch(e){}

}

const sfx={click:()=>beep(500,0.08,'triangle',0.1),ok:()=>{beep(600,0.12);setTimeout(()=>beep(900,0.15),100);},bad:()=>beep(150,0.3,'sawtooth',0.15),done:()=>{beep(500,0.15);setTimeout(()=>beep(700,0.15),150);setTimeout(()=>beep(1000,0.3),300);}};


function startGame(){

    state={idx:0,score:0,answered:false};

    D.start.classList.remove('active');

    D.results.classList.remove('active');

    D.play.classList.add('active');

    D.hudScore.textContent=0;

    loadQ();

}


function loadQ(){

    if(state.idx>=questions.length){return endGame();}

    state.answered=false;

    const q=questions[state.idx];

    D.organEmoji.textContent=q.emoji;

    D.organName.textContent=q.name;

    D.hudQ.textContent=state.idx+1;

    D.prog.style.width=`${(state.idx/questions.length)*100}%`;

    D.options.innerHTML='';

    D.feedback.className='feedback';

    D.nextBtn.classList.remove('show');


    q.options.forEach((opt,i)=>{

        const el=document.createElement('div');

        el.className='option';

        el.textContent=opt;

        el.addEventListener('click',()=>answer(i));

        D.options.appendChild(el);

    });

}


function answer(i){

    if(state.answered) return;

    state.answered=true;

    const q=questions[state.idx];

    const isRight=i===q.correct;


    document.querySelectorAll('.option').forEach((o,idx)=>{

        o.style.pointerEvents='none';

        if(idx===q.correct) o.classList.add('correct');

        else if(idx===i) o.classList.add('wrong');

    });


    if(isRight){

        sfx.ok(); state.score++;

        D.hudScore.textContent=state.score;

        D.feedback.className='feedback show ok';

        D.feedback.innerHTML=`✅ <strong>Correct!</strong> ${q.explain}`;

    }else{

        sfx.bad();

        D.feedback.className='feedback show bad';

        D.feedback.innerHTML=`❌ <strong>Not quite.</strong> ${q.explain}`;

    }


    D.nextBtn.classList.add('show');

    D.nextBtn.textContent=state.idx<questions.length-1?'Next →':'See Results →';

}


D.nextBtn.addEventListener('click',()=>{sfx.click();state.idx++;if(state.idx>=questions.length)endGame();else loadQ();});


function endGame(){

    D.play.classList.remove('active');

    D.results.classList.add('active');

    D.prog.style.width='100%';

    D.finalScore.textContent=`${state.score}/${questions.length}`;

    const pct=state.score/questions.length;

    let badge,title,msg;

    if(pct>=0.85){badge='🏆';title='System Expert!';msg='You know all the major animal organ systems! Excellent anatomical knowledge.';}

    else if(pct>=0.6){badge='🫀';title='Anatomy Adept';msg='You recognize most systems. Review the ones you missed to master this concept.';}

    else{badge='🌱';title='Growing Anatomist';msg='Keep exploring! Each organ has a specific system it belongs to — practice makes perfect.';}

    D.levelBadge.textContent=badge;

    D.levelTitle.textContent=title;

    D.resultsMsg.textContent=msg;

    sfx.done();

}


D.startBtn.addEventListener('click',()=>{sfx.click();startGame();});

D.restartBtn.addEventListener('click',()=>{sfx.click();startGame();});
